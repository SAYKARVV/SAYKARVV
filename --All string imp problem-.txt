        void String6()
        {
            int i, j, cnt;
            string str = "hellllo";
            string msg = " ";
            for (i = 0; i < str.Length; i++)
            {
                cnt = 0;
                for (j = 0; j < msg.Length; j++)
                {
                    if (str[i] == msg[j])
                    {
                        cnt++;
                        break;
                    }
                }
                if (cnt == 0)
                {
                    msg = msg + str[i];
                }
            }
            Console.WriteLine(msg);
        }
        //---------------------Find palindrome string or not-------------

      void String7()
        {
            int i, cnt = 0;
            string str = "madama";
            int n = str.Length - 1;

            for (i = 0; i < str.Length/2; i++)
            {
                if (str[i] != str[n])
                {
                    cnt++;
                    break;
                }
                n--;
            }
            if (cnt == 0)
            {
                Console.WriteLine("Palindrome");
            }
            else
            {
                Console.WriteLine("Not Palindrome");
            }
        }

        void String8()
        {
            string str;
            char ch;
            int i, cnt = 0;
            Console.WriteLine("Enter any string");
            str = Console.ReadLine();

            Console.WriteLine("Enter any character");
            ch = Convert.ToChar(Console.ReadLine());

            int p = (int)ch;
            if (p >= 97)
            {
                p = p - 32;
            }
            ch = (char)p;
            for (i = 0; i < str.Length; i++)
            {
                int d = (int)str[i];
                if (d >= 97)
                {
                    d = d - 32;
                }
                char c= (char)d;
                if (ch == c)
                {
                    cnt++;
                    break;
                }
            }
            Console.WriteLine(cnt);
        }
