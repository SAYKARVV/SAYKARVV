    //---------------------------------------* patern pyramid ---------------------------------

        void Pyramid()
        {
            int i, j, k,n;
            Console.WriteLine("Enter any number:");
            n = Convert.ToInt32(Console.ReadLine());

            for (i = 1; i <= n; i++)
            {
                for (j = 1; j <= n - i; j++)
                {
                    Console.Write(" ");
                }
                for (k = i; k >= 1; k--)
                {
                    Console.Write("*");
                }
                for (k = 2; k <= i; k++)
                {
                    Console.WriteLine("*");
                }
                Console.WriteLine();
            }
            
        }

    //------------------------while loop using Reverese Number----------------------------------

        void ReverseNumber()
        {
            int n = 123, rem = 0, sum = 0;
            while (n > 0)
            {
                rem = n % 10;
                sum = rem + (sum * 10);
                n = n / 10;
            }
            Console.WriteLine("reverse number:" + sum);
        }

        //------------------------decimal(1 to 9) to binary(0,1) convert-----------
        void DecimaltoBi()
        {
            int n, rem; string str = " ";
            Console.WriteLine("Enter Decimal Number:");
            n = Convert.ToInt32(Console.ReadLine());

            while (n != 0)
            {
                rem = n % 2;
                str = rem + str;
                n = n / 2;
            }
            Console.WriteLine("binary number is:" + str);
        }
        //-------------------------binary to decimal-------------------------

        void Binarytodec()
        {
            int n, rem, sum = 0, p = 0;
            Console.WriteLine("enter any binary num:");
            n = Convert.ToInt32(Console.ReadLine());

            while (n != 0)
            {
                rem = n % 10;
                sum = sum + rem * (int)Math.Pow(2, p);
                n = n / 10;
                p++;
            }
            Console.WriteLine("Decimal no is:" + sum);
        }
