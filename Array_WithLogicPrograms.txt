		void Armstrong()
		{
			int n, r, sum = 0, temp;
			Console.WriteLine("Enter the number:");
			n = int.Parse(Console.ReadLine());
			temp = n;
			while (n > 0)
			{
				r = n % 10;
				sum = sum + (r * r * r);
				n = n / 10;
			}
			if (temp == sum)
			{
				Console.WriteLine("armstrong number");
			}
			else
			{
				Console.WriteLine("Not armstrong number");
			}
		}

		void Decimaltobinary()//start 1,2,4,8
		{
			int n, rem;
			string str = "";
			Console.WriteLine("Enter the decimal number");
			n = Convert.ToInt32(Console.ReadLine());

			while (n > 0)

				rem = n % 2;
				str = rem + str;
				n = n / 2;
			}
			Console.WriteLine("Binary Number is:" + str);
		}
//-----------------------binary to decimal----------------------------

		void Binarytodecimal()
		{
			int n, rem, sum = 0, p = 0;
			Console.WriteLine("Enter binary number");
			n= Convert.ToInt32(Console.ReadLine()); 

			while (n > 0)
			{
				rem = n % 10;
				sum = sum + rem * (int)Math.Pow(2, p);
				n = n / 10;
				p++;
			}
			Console.WriteLine("Decimal num is:" + sum);
		}
